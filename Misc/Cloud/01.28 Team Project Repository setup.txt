(GPT가 고정(hard-coding) ID(deploy) 생성 스크립트를 만들어준 것을 모르고 실행해서
다음 명령어로 삭제했음: sudo deluser --remove-home deploy)

사용자 계정명을 인자로 받는 script:
1. Script
#!/bin/bash

# 인자 체크
if [ -z "$1" ]; then
  echo "사용법: $0 <사용자계정명>"
  exit 1
fi

NEW_USER="$1"
COPY_SSH_KEYS=true

set -e

echo "▶ 사용자 생성: $NEW_USER"

sudo adduser --disabled-password --gecos "" "$NEW_USER"
sudo usermod -aG sudo "$NEW_USER"

if [ "$COPY_SSH_KEYS" = true ]; then
  echo "▶ SSH 키 복사 중..."
  sudo mkdir -p /home/$NEW_USER/.ssh
  sudo cp /home/ubuntu/.ssh/authorized_keys /home/$NEW_USER/.ssh/
  sudo chown -R $NEW_USER:$NEW_USER /home/$NEW_USER/.ssh
  sudo chmod 700 /home/$NEW_USER/.ssh
  sudo chmod 600 /home/$NEW_USER/.ssh/authorized_keys
fi

echo "✅ 완료! 접속: ssh $NEW_USER@<EC2_PUBLIC_IP>"


2. Script 사용법(생성 방법)

* 아래는 강사님 스크립트 (유료 플랜이라 그런지 엄청 상세하게 만들어줬다)
"#!/usr/bin/env bash
# EC2에서 신규 사용자 계정을 만들고 공개키를 등록하는 스크립트
# 사용법:
#   sudo ./create_ec2_user.sh USERNAME ""ssh-ed25519 AAAA.... user@example""
#
# 주의: 전체 공개키 문자열을 큰따옴표("")로 감싸서 한 번에 전달해야 합니다.

set -euo pipefail

### 설정 영역 ###
# 새 사용자를 sudo 그룹에 넣고 싶으면 true, 아니면 false
ADD_TO_SUDO=true
###############

if [[ ""$EUID"" -ne 0 ]]; then
  echo ""이 스크립트는 root 또는 sudo로 실행해야 합니다."" >&2
  exit 1
fi

if [[ $# -lt 2 ]]; then
  echo ""사용법: $0 USERNAME \""SSH_PUBLIC_KEY\"""" >&2
  echo ""예시 : $0 devuser \""ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAI... devuser@pc\"""" >&2
  exit 1
fi

USERNAME=""$1""
PUBKEY=""$2""

echo ""▶ 새 사용자 생성/설정 시작: ${USERNAME}""

# 1. 사용자 존재 여부 확인
if id ""$USERNAME"" >/dev/null 2>&1; then
  echo ""ℹ️  사용자 '${USERNAME}'는 이미 존재합니다. 홈 디렉토리와 SSH 키만 설정합니다.""
else
  echo ""1) 사용자 '${USERNAME}' 생성 중...""
  # -m: 홈 디렉토리 생성, -s: 로그인 셸 지정
  useradd -m -s /bin/bash ""$USERNAME""

  if [[ ""$ADD_TO_SUDO"" == ""true"" ]]; then
    echo ""   '${USERNAME}'를 sudo 그룹에 추가합니다.""
    usermod -aG sudo ""$USERNAME""
  fi
fi

# 2. 홈 디렉토리와 .ssh 디렉토리 준비
USER_HOME=$(eval echo ""~${USERNAME}"")

if [[ ! -d ""$USER_HOME"" ]]; then
  echo ""홈 디렉토리가 없어서 새로 생성합니다: $USER_HOME""
  mkdir -p ""$USER_HOME""
  chown ""$USERNAME:$USERNAME"" ""$USER_HOME""
fi

SSH_DIR=""${USER_HOME}/.ssh""
AUTH_KEYS=""${SSH_DIR}/authorized_keys""

echo ""2) ${SSH_DIR} 디렉토리 및 authorized_keys 설정...""

mkdir -p ""$SSH_DIR""
chmod 700 ""$SSH_DIR""
chown ""$USERNAME:$USERNAME"" ""$SSH_DIR""

# 3. authorized_keys에 공개키 등록
#    이미 파일이 있으면 맨 아래에 추가합니다.
if [[ -f ""$AUTH_KEYS"" ]]; then
  echo ""   기존 authorized_keys가 있습니다. 공개키를 추가합니다.""
else
  echo ""   새 authorized_keys 파일을 생성합니다.""
  touch ""$AUTH_KEYS""
fi

# 같은 키가 이미 있는지 대략적으로 체크 (중복 추가 방지)
if grep -Fq ""$PUBKEY"" ""$AUTH_KEYS""; then
  echo ""   이미 동일한 공개키가 등록되어 있습니다. 추가하지 않습니다.""
else
  echo ""$PUBKEY"" >> ""$AUTH_KEYS""
  echo ""   공개키를 authorized_keys에 추가했습니다.""
fi

chmod 600 ""$AUTH_KEYS""
chown ""$USERNAME:$USERNAME"" ""$AUTH_KEYS""

echo ""3) 설정 요약""
echo ""   - 사용자명   : $USERNAME""
echo ""   - 홈 디렉토리: $USER_HOME""
echo ""   - SSH 디렉토리: $SSH_DIR""
echo ""   - authorized_keys: $AUTH_KEYS""
if [[ ""$ADD_TO_SUDO"" == ""true"" ]]; then
  echo ""   - sudo 그룹  : 추가됨""
else
  echo ""   - sudo 그룹  : 추가 안 함""
fi

echo ""✅ 완료! 이제 다음과 같이 접속을 테스트해 보세요:""
echo ""   ssh -i /path/to/private_key ${USERNAME}@<EC2_PUBLIC_IP_OR_DNS>""
"
